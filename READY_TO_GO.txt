🎓 KEY CODE SNIPPETS READY TO USE

Generate Seed:
import { generateMnemonic } from './lib/bip39';
const seed = generateMnemonic(12);  // "abandon able zone ..."

Encode to Cloak:
import { encodeWithCipher } from './lib/cipher';
const cloak = encodeWithCipher(seed, wordlist, cipher);

Decode Cloak:
import { decodeWithCipher } from './lib/cipher';
const realSeed = decodeWithCipher(cloak, wordlist, cipher);

Derive Wallets:
import { deriveWalletAddresses } from './lib/bip39';
const addresses = deriveWalletAddresses(seed);
// { ethereum: '0x...', bitcoin: 'bc1q...', solana: 'Sol...' }

Save Cloak:
import { addCloakSeed } from './lib/storage';
const id = addCloakSeed('My Cloak', 'animals', cipher);
